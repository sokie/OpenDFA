cmake_minimum_required(VERSION 3.15)
project(opendfa-minimal C)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "OpenDFA must be built 32-bit. Configure with -A Win32.")
endif()
add_library(dfa SHARED src/des.c src/g.c src/dfa.c dfa.def)
set_target_properties(dfa PROPERTIES OUTPUT_NAME "DFA" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin")
target_include_directories(dfa PRIVATE src)
set_property(TARGET dfa PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
if(MSVC)
    target_compile_options(dfa PRIVATE /MT)
endif()
